import os
from dotenv import load_dotenv
from psycopg import sql
from psycopg.rows import dict_row
from psycopg_pool import ConnectionPool

load_dotenv()

SCHEMA = os.getenv("DB_SCHEMA", "IntroToSoftware_group4Project")


def _use_schema(conn):
    # Runs once for every new connection the pool opens.
    conn.execute(sql.SQL("SET search_path TO {}, public").format(sql.Identifier(SCHEMA)))
    conn.commit()


pool = ConnectionPool(
    os.environ["DATABASE_URL"],
    min_size=1,
    max_size=10,
    kwargs={"row_factory": dict_row},
    configure=_use_schema,
    open=True,
)


def fetch_all(sql_text, params=()):
    with pool.connection() as conn:
        return conn.execute(sql_text, params).fetchall()


def fetch_one(sql_text, params=()):
    with pool.connection() as conn:
        return conn.execute(sql_text, params).fetchone()

"""Per-user queries. Every function returns data shaped like sampleData.js."""
import os
from datetime import date, datetime, timedelta
from zoneinfo import ZoneInfo

from db import fetch_all, fetch_one, pool

TZNAME = os.getenv("APP_TIMEZONE", "America/Denver")
TZ = ZoneInfo(TZNAME)


# ---------- label helpers ----------
def _local(dt: datetime) -> datetime:
    return dt.astimezone(TZ)


def _time_label(dt: datetime) -> str:
    return _local(dt).strftime("%I:%M %p").lstrip("0")  # "10:00 AM"


def _due_label(dt: datetime, today: date) -> str:
    local = _local(dt)
    if local.date() == today:
        return f"Today, {_time_label(dt)}"
    return f"{local.strftime('%a, %b')} {local.day}"  # "Wed, Sep 23"


def _updated_label(dt: datetime) -> str:
    local = _local(dt)
    return f"{local.strftime('%b')} {local.day}, {_time_label(dt)}"


def week_bounds(today: date):
    start = today - timedelta(days=today.weekday())  # Monday
    end = start + timedelta(days=6)
    if start.month == end.month:
        label = f"{start:%B} {start.day}–{end.day}, {end.year}"
    else:
        label = f"{start:%B} {start.day} – {end:%B} {end.day}, {end.year}"
    return {
        "label": label,
        "todayLabel": f"{today:%A, %B} {today.day}",
        "startDate": start.isoformat(),
        "endDate": end.isoformat(),
    }


# ---------- queries ----------
def get_user(user_id):
    row = fetch_one("SELECT user_name FROM users WHERE id = %s", (user_id,))
    if not row:
        return None
    name = row["user_name"]
    parts = name.split()
    initials = (parts[0][0] + parts[-1][0]) if len(parts) > 1 else name[:2]
    return {"name": name, "initials": initials.upper()}


def get_classes(user_id):
    rows = fetch_all(
        "SELECT slug, course_name, color FROM courses WHERE user_id = %s ORDER BY id",
        (user_id,),
    )
    return [{"id": r["slug"], "name": r["course_name"], "color": r["color"]} for r in rows]


def get_assignments(user_id, today=None):
    today = today or datetime.now(TZ).date()
    rows = fetch_all(
        """
        SELECT e.id, e.event_name, e.detail, e.starts_at, a.status, c.slug, c.id AS course_id, a.days_before_alert
        FROM assignments a
        JOIN events e ON e.id = a.event_id
        LEFT JOIN courses c ON c.id = e.course_id
        WHERE e.user_id = %s
        ORDER BY e.starts_at
        """,
        (user_id,),
    )
    return [
        {
            "id": str(r["id"]),
            "dueDate": _local(r["starts_at"]).date().isoformat(),
            "title": r["event_name"],
            "detail": r["detail"],
            "dueLabel": _due_label(r["starts_at"], today),
            "status": r["status"],
            "classId": r["slug"],
            "courseId": r["course_id"],
            "daysBeforeAlert": r["days_before_alert"],
        }
        for r in rows
    ]


def get_notes(user_id, limit=None):
    sql = """
        SELECT n.id, n.title, n.contents, n.pinned, n.course_id, n.updated_at, c.slug
        FROM notes n LEFT JOIN courses c ON c.id = n.course_id
        WHERE n.user_id = %s ORDER BY n.updated_at DESC
    """
    params = [user_id]
    if limit:
        sql += " LIMIT %s"
        params.append(limit)
    return [
        {
            "id": str(r["id"]),
            "title": r["title"],
            "contents": r["contents"],
            "pinned": r["pinned"],
            "classId": r["slug"],
            "updated": _local(r["updated_at"]).date().isoformat(),
            "updatedLabel": _updated_label(r["updated_at"]),
        }
        for r in fetch_all(sql, params)
    ]


def get_upcoming(user_id, day=None):
    """Everything happening on one local day (default: today), non-campus events."""
    day = day or datetime.now(TZ).date()
    rows = fetch_all(
        """
        SELECT e.id, e.event_name, e.detail, e.starts_at, c.slug,
               (a.event_id IS NOT NULL) AS is_assignment
        FROM events e
        LEFT JOIN assignments a ON a.event_id = e.id
        LEFT JOIN courses c ON c.id = e.course_id
        WHERE e.user_id = %s AND NOT e.school_event
          AND (e.starts_at AT TIME ZONE %s)::date = %s
        ORDER BY e.starts_at
        """,
        (user_id, TZNAME, day),
    )
    return [
        {
            "id": str(r["id"]),
            "time": _time_label(r["starts_at"]),
            "title": r["event_name"],
            "detail": r["detail"],
            "classId": r["slug"],
            "type": "Due today" if r["is_assignment"] else "Class",
        }
        for r in rows
    ]


def get_campus_events(user_id):
    rows = fetch_all(
        """
        SELECT id, event_name, starts_at FROM events
        WHERE user_id = %s AND school_event ORDER BY starts_at
        """,
        (user_id,),
    )
    return [
        {"id": str(r["id"]), "title": r["event_name"], "startsAt": r["starts_at"].isoformat()}
        for r in rows
    ]


def get_home(user_id, today=None):
    """Everything App.jsx currently reads from sampleData.js, in one call."""
    today = today or datetime.now(TZ).date()
    return {
        "user": get_user(user_id),
        "classes": get_classes(user_id),
        "week": week_bounds(today),
        "upcoming": get_upcoming(user_id, today),
        "assignments": get_assignments(user_id, today),
        "campusEvents": get_campus_events(user_id),
        "notes": get_notes(user_id),
    }

def _course_id(conn, user_id, d):
    """Accepts a numeric courseId (app.js) or a slug classId (React)."""
    if d.get("courseId"):
        row = conn.execute("SELECT id FROM courses WHERE id = %s AND user_id = %s",
                           (d["courseId"], user_id)).fetchone()
    elif d.get("classId"):
        row = conn.execute("SELECT id FROM courses WHERE slug = %s AND user_id = %s",
                           (d["classId"], user_id)).fetchone()
    else:
        return None
    return row["id"] if row else None


def create_assignment(user_id, d):
    with pool.connection() as conn:
        ev = conn.execute(
            """INSERT INTO events (user_id, course_id, event_name, detail, starts_at)
               VALUES (%s, %s, %s, %s, (%s::date + time '23:59') AT TIME ZONE %s)
               RETURNING id""",
            (user_id, _course_id(conn, user_id, d), d["title"], d.get("detail", ""),
             d["dueDate"], TZNAME),
        ).fetchone()
        conn.execute(
            "INSERT INTO assignments (event_id, days_before_alert, status) VALUES (%s, %s, %s)",
            (ev["id"], int(d.get("daysBeforeAlert", 1)), d.get("status", "Not started")),
        )
    return ev["id"]


def update_assignment(user_id, event_id, ch):
    with pool.connection() as conn:
        owned = conn.execute(
            """SELECT 1 FROM events e JOIN assignments a ON a.event_id = e.id
               WHERE e.id = %s AND e.user_id = %s""", (event_id, user_id)).fetchone()
        if not owned:
            return False
        if "status" in ch:
            conn.execute("UPDATE assignments SET status = %s WHERE event_id = %s",
                         (ch["status"], event_id))
        if "daysBeforeAlert" in ch:
            conn.execute("UPDATE assignments SET days_before_alert = %s WHERE event_id = %s",
                         (int(ch["daysBeforeAlert"]), event_id))
        if "title" in ch:
            conn.execute("UPDATE events SET event_name = %s WHERE id = %s", (ch["title"], event_id))
        if "detail" in ch:
            conn.execute("UPDATE events SET detail = %s WHERE id = %s", (ch["detail"], event_id))
        if "dueDate" in ch:
            conn.execute(
                "UPDATE events SET starts_at = (%s::date + time '23:59') AT TIME ZONE %s WHERE id = %s",
                (ch["dueDate"], TZNAME, event_id))
        if "courseId" in ch or "classId" in ch:
            conn.execute("UPDATE events SET course_id = %s WHERE id = %s",
                         (_course_id(conn, user_id, ch), event_id))
    return True


def delete_assignment(user_id, event_id):
    with pool.connection() as conn:
        cur = conn.execute(
            """DELETE FROM events WHERE id = %s AND user_id = %s
               AND id IN (SELECT event_id FROM assignments)""", (event_id, user_id))
        return cur.rowcount > 0
    

def _owned_course(conn, user_id, course_id):
    if not course_id:
        return None
    row = conn.execute(
        "SELECT id FROM courses WHERE id = %s AND user_id = %s", (course_id, user_id)
    ).fetchone()
    return row["id"] if row else None


def create_note(user_id, title, contents, course_id, pinned=False):
    with pool.connection() as conn:
        row = conn.execute(
            """INSERT INTO notes (user_id, course_id, title, contents, pinned)
               VALUES (%s, %s, %s, %s, %s) RETURNING id""",
            (user_id, _owned_course(conn, user_id, course_id), title, contents, pinned),
        ).fetchone()
    return row["id"]


def update_note(user_id, note_id, d):
    with pool.connection() as conn:
        cur = conn.execute(
            """UPDATE notes SET title = %s, contents = %s, course_id = %s,
                      pinned = %s, updated_at = now()
               WHERE id = %s AND user_id = %s""",
            (d["title"], d.get("contents", ""), _owned_course(conn, user_id, d.get("courseId")),
             bool(d.get("pinned")), note_id, user_id),
        )
        return cur.rowcount > 0

def delete_note(user_id, note_id):
    with pool.connection() as conn:
        cur = conn.execute("DELETE FROM notes WHERE id = %s AND user_id = %s", (note_id, user_id))
        return cur.rowcount > 0

"""The API (architecture Layer 3): four Flask blueprints, JSON only.

    auth_bp       auth.py              -> AuthService
    notes_bp      notes.py             -> NoteRouter
    calendar_bp   calendar.py, home.py -> ScheduleService, ReminderService
    resources_bp  resources.py         -> ResourceService, ScheduleService

Layers below: services/ (business logic), models.py + queries.py (data access),
db.py (connection pool) -> PostgreSQL.

Every route answers at /api/... (as before) and at /api/v1/... (architecture).
Log in with the bearer token (Authorization: Bearer <token>), as before, or
with the session cookie + X-CSRF-Token header (see session.py).
Endpoint reference: README.md in this folder.
"""
import logging
import os
import secrets

from flask import Blueprint, abort, current_app, g, jsonify, request
from psycopg import OperationalError, errors as pg_errors
from psycopg_pool import PoolTimeout
from werkzeug.exceptions import HTTPException

from api.errors import DomainError

api = Blueprint("api", __name__, url_prefix="/api")
v1 = Blueprint("v1", __name__, url_prefix="/v1")
log = logging.getLogger(__name__)

HTTP_ERRORS = {
    400: "bad_request",
    401: "unauthorized",
    403: "csrf_failed",
    404: "not_found",
    405: "method_not_allowed",
    409: "conflict",
    415: "unsupported_media_type",
}


@api.record_once
def _configure_sessions(state):
    """Sessions need a secret key; app.py doesn't set one, so the API does."""
    app = state.app
    if not app.secret_key:
        app.secret_key = os.environ.get("SECRET_KEY") or secrets.token_hex(32)
    if app.config.get("SESSION_COOKIE_SAMESITE") is None:
        app.config["SESSION_COOKIE_SAMESITE"] = "Lax"  # httpOnly is Flask's default


@api.before_request
def authenticate():
    """Bearer token (as before) or session cookie + CSRF token. Sets g.user_id."""
    from api.services import auth_service
    from api.session import csrf_ok, session_user

    view = current_app.view_functions.get(request.endpoint)
    if getattr(view, "is_public", False):
        return
    header = request.headers.get("Authorization", "")
    if header.startswith("Bearer "):
        user_id = auth_service.user_for_token(header[7:])
        if user_id is None:
            abort(401)
        g.user_id, g.token = user_id, header[7:]
        return
    user_id = session_user()
    if header or user_id is None:  # another Authorization scheme, or not logged in
        abort(401)
    if not csrf_ok():
        abort(403, "Send the X-CSRF-Token header you received at login.")
    g.user_id, g.token = user_id, None


# ---------- errors: always JSON {"error": code, "message": text} ----------
def error(status, code, message, **extra):
    return jsonify(error=code, message=message, **extra), status


def _http_error_json(e: HTTPException):
    if e.code == 401:
        return jsonify(error="unauthorized"), 401  # same reply as the original api.py
    custom = e.description != type(e).description  # abort(403, "...") gave its own text
    return error(e.code, HTTP_ERRORS.get(e.code, "error"), e.description if custom else e.name)


@api.errorhandler(Exception)
def _handle_error(e):
    if isinstance(e, DomainError):  # typed exceptions from the service layer
        extra = {"fields": e.fields} if e.fields else {}
        return error(e.status, e.code, e.message, **extra)
    if isinstance(e, HTTPException):
        return _http_error_json(e)
    if isinstance(e, pg_errors.NumericValueOutOfRange):  # an id too big to exist
        return error(404, "not_found", "Not found.")
    if isinstance(e, (OperationalError, PoolTimeout)):
        log.exception("Database unavailable")
        return error(503, "database_unavailable", "The database is not reachable. Is PostgreSQL running?")
    log.exception("Unhandled API error")
    return error(500, "server_error", "Something went wrong on the server.")


@api.app_errorhandler(404)
@api.app_errorhandler(405)
def _unmatched_api_route(e):
    """Unknown /api/... URLs never reach a blueprint, so answer them app-wide in JSON."""
    return _http_error_json(e) if request.path.startswith("/api/") else e

"""auth_bp: register, login, logout, current user. Calls AuthService."""
from flask import Blueprint, g, jsonify

from api.helpers import body, ok, public
from api.services import auth_service
from api.session import end_session, start_session
from api.validation import Fields

auth_bp = Blueprint("auth_bp", __name__)


@auth_bp.post("/auth/register")
@public
def register():
    f = Fields(body())
    f.text("name", required=True, max_len=80)
    f.email("email", required=True)
    f.password("password", required=True)
    v = f.done()
    user_id, token = auth_service.register(v["name"], v["email"], v["password"])
    return jsonify(token=token, user=auth_service.get_user(user_id), csrfToken=start_session(user_id)), 201


@auth_bp.post("/auth/login")
@public
def login():
    f = Fields(body())
    f.email("email", required=True)
    f.text("password", required=True, max_len=200, strip=False)
    v = f.done()
    user_id, token = auth_service.login(v["email"], v["password"])
    return jsonify(token=token, user=auth_service.get_user(user_id), csrfToken=start_session(user_id))


@auth_bp.post("/auth/logout")
def logout():
    if g.token:
        auth_service.logout(g.token)
    end_session()
    return ok()


@auth_bp.get("/me")
def me():
    return jsonify(auth_service.get_user(g.user_id))

"""Typed domain exceptions (architecture: "in-process method calls, typed domain exceptions").

Services raise these and know nothing about HTTP. api/__init__.py turns each one
into its status code and the JSON body {"error": code, "message": text}.
"""


class DomainError(Exception):
    status = 400
    code = "error"

    def __init__(self, message, fields=None):
        super().__init__(message)
        self.message = message
        self.fields = fields


class ValidationError(DomainError):
    """Bad input. `fields` maps each field name to what is wrong with it."""

    status = 400
    code = "validation_failed"

    def __init__(self, fields):
        super().__init__("Some fields are missing or invalid.", fields)


class NotFound(DomainError):
    """The item doesn't exist or belongs to another user (deliberately indistinguishable)."""

    status = 404
    code = "not_found"

    def __init__(self, what):
        super().__init__(f"{what} not found.")


class InvalidCredentials(DomainError):
    status = 401
    code = "invalid_credentials"

    def __init__(self):
        super().__init__("Email or password is incorrect.")


class AlreadyExists(DomainError):
    """A unique value (such as an email address) is already taken."""

    status = 409

    def __init__(self, code, message):
        super().__init__(message)
        self.code = code

FILE: code/api/session.py
LANGUAGE: Python
PURPOSE: httpOnly session cookie and CSRF token handling.
===============================================================================

"""Session cookie + CSRF token (architecture: "httpOnly session cookie, CSRF token").

Login and register now also start a session: Flask's signed, httpOnly cookie
holds the user id, and the JSON reply includes a `csrfToken`. A browser client
can then skip the bearer token and instead send the cookie automatically plus
the header X-CSRF-Token on every POST/PATCH/DELETE.

The bearer token (Authorization: Bearer ...) keeps working exactly as before,
so existing clients are unaffected.
"""
import hmac
import secrets

from flask import request, session

CSRF_HEADER = "X-CSRF-Token"
SAFE_METHODS = frozenset({"GET", "HEAD", "OPTIONS"})


def start_session(user_id):
    session.clear()  # a fresh session on every login
    session.permanent = True
    session["user_id"] = user_id
    session["csrf"] = secrets.token_urlsafe(32)
    return session["csrf"]


def end_session():
    session.clear()


def session_user():
    return session.get("user_id")


def csrf_ok():
    """Writes made with the cookie must echo the CSRF token from login."""
    if request.method in SAFE_METHODS:
        return True
    sent = request.headers.get(CSRF_HEADER, "")
    return bool(sent) and hmac.compare_digest(sent, session.get("csrf", ""))

FILE: code/api/validation.py
LANGUAGE: Python
PURPOSE: Request field validation (types, formats, lengths, ranges).
===============================================================================

"""Request validation for the API.

    f = Fields(body())                    # partial=True for PATCH
    f.text("title", required=True, max_len=100)
    f.date("dueDate", required=True)
    values = f.done()                     # raises ValidationError listing every bad field

Only keys that were sent end up in `values`, so PATCH can tell "not sent"
apart from "sent as null" (nullable fields keep None).
"""
import re
from datetime import date, time

from api.errors import ValidationError

WEEKDAYS = ("Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun")
STATUSES = ("Not started", "In progress", "Completed")

_DATE_RE = re.compile(r"^\d{4}-\d{2}-\d{2}$")
_TIME_RE = re.compile(r"^\d{2}:\d{2}$")
_EMAIL_RE = re.compile(r"^[^@\s]+@[^@\s]+\.[^@\s]+$")
_COLOR_RE = re.compile(r"^#[0-9a-fA-F]{6}$")
_MISSING = object()


def parse_date(raw, field):
    if not isinstance(raw, str) or not _DATE_RE.match(raw):
        raise ValidationError({field: "Use the format YYYY-MM-DD."})
    try:
        return date.fromisoformat(raw)
    except ValueError:
        raise ValidationError({field: "Not a real calendar date."}) from None


def parse_time(raw, field):
    if not isinstance(raw, str) or not _TIME_RE.match(raw):
        raise ValidationError({field: "Use 24-hour HH:MM, e.g. 09:30."})
    try:
        return time.fromisoformat(raw)
    except ValueError:
        raise ValidationError({field: "Not a real time of day."}) from None


class Fields:
    def __init__(self, data, *, partial=False):
        if not isinstance(data, dict):
            raise ValidationError({"body": "Send a JSON object with Content-Type: application/json."})
        self._data = data
        self._partial = partial
        self.errors = {}
        self.values = {}

    def _take(self, key, required, nullable):
        if key not in self._data:
            if required and not self._partial:
                self.errors[key] = "This field is required."
            return _MISSING
        value = self._data[key]
        if value is None:
            if nullable:
                self.values[key] = None
            else:
                self.errors[key] = "This field is required." if required else "Must not be null."
            return _MISSING
        return value

    def text(self, key, *, required=False, max_len=200, nullable=False, allow_blank=False, strip=True):
        value = self._take(key, required, nullable)
        if value is _MISSING:
            return
        if not isinstance(value, str):
            self.errors[key] = "Must be a string."
            return
        if strip:
            value = value.strip()
        if not value and not allow_blank:
            self.errors[key] = "Must not be blank."
        elif len(value) > max_len:
            self.errors[key] = f"Must be at most {max_len} characters."
        else:
            self.values[key] = value

    def email(self, key, *, required=False):
        self.text(key, required=required, max_len=254)
        if key in self.values:
            if _EMAIL_RE.match(self.values[key]):
                self.values[key] = self.values[key].lower()
            else:
                del self.values[key]
                self.errors[key] = "Enter a valid email address."

    def password(self, key, *, required=False):
        self.text(key, required=required, max_len=200, strip=False)
        if key in self.values and len(self.values[key]) < 8:
            del self.values[key]
            self.errors[key] = "Must be at least 8 characters."

    def date(self, key, *, required=False):
        value = self._take(key, required, False)
        if value is not _MISSING:
            try:
                self.values[key] = parse_date(value, key)
            except ValidationError as e:
                self.errors.update(e.fields)

    def time(self, key, *, required=False, nullable=False):
        value = self._take(key, required, nullable)
        if value is not _MISSING:
            try:
                self.values[key] = parse_time(value, key)
            except ValidationError as e:
                self.errors.update(e.fields)

    def choice(self, key, options, *, required=False):
        value = self._take(key, required, False)
        if value is _MISSING:
            return
        if value in options:
            self.values[key] = value
        else:
            self.errors[key] = "Must be one of: " + ", ".join(options) + "."

    def integer(self, key, *, lo, hi, required=False, nullable=False):
        value = self._take(key, required, nullable)
        if value is _MISSING:
            return
        if isinstance(value, bool) or not isinstance(value, int):
            self.errors[key] = "Must be a whole number."
        elif not lo <= value <= hi:
            self.errors[key] = f"Must be between {lo} and {hi}."
        else:
            self.values[key] = value

    def boolean(self, key, *, required=False):
        value = self._take(key, required, False)
        if value is _MISSING:
            return
        if isinstance(value, bool):
            self.values[key] = value
        else:
            self.errors[key] = "Must be true or false."

    def color(self, key):
        value = self._take(key, False, False)
        if value is _MISSING:
            return
        if isinstance(value, str) and _COLOR_RE.match(value):
            self.values[key] = value.lower()
        else:
            self.errors[key] = "Use a hex color like #e3d6fa."

    def weekdays(self, key, *, max_items):
        """A list of distinct day names (Mon..Sun). null is treated as []."""
        value = self._take(key, False, True)
        if value is _MISSING:
            if key in self.values:  # sent as null
                self.values[key] = []
            return
        if not isinstance(value, list) or any(d not in WEEKDAYS for d in value):
            self.errors[key] = "Must be a list of days from: " + ", ".join(WEEKDAYS) + "."
        elif len(set(value)) != len(value):
            self.errors[key] = "Each day may appear only once."
        elif len(value) > max_items:
            self.errors[key] = f"At most {max_items} days."
        else:
            self.values[key] = value

    def class_ref(self):
        """Which class an item belongs to: `classId` (slug, used by React) or
        `courseId` (number, used by the templates/ prototype). null = no class."""
        self.text("classId", max_len=60, nullable=True)
        self.integer("courseId", lo=1, hi=2_147_483_647, nullable=True)

    def done(self):
        if self.errors:
            raise ValidationError(self.errors)
        if self._partial and not self.values:
            raise ValidationError({"body": "Send at least one field to update."})
        return self.values


def query_date(args, key):
    raw = args.get(key)
    return parse_date(raw, key) if raw else None


def changes_class(values):
    """True if the request sent a class (classId or courseId), even as null."""
    return "classId" in values or "courseId" in values

"""Data access layer (architecture Layer 5): User, Token, Course, Event, Assignment, Note.

All SQL written for the API lives here. Every method takes an open connection,
so a service can run several steps inside one transaction. Each model also
knows its JSON shape (`to_json`). The read queries used by the original screens
stay in queries.py (Lee's) and are reused as they are.

import re
import secrets
from datetime import datetime, time

from psycopg import sql

from api.errors import ValidationError
from queries import TZ, TZNAME, _due_label, _local, _time_label, _updated_label


def _hhmm(value):
    return value.strftime("%H:%M") if value is not None else None


def at_local(day, at: time) -> datetime:
    """A date + time of day in the app's timezone."""
    return datetime.combine(day, at, tzinfo=TZ)


def clock(dt: datetime) -> time:
    """The local time of day of a stored timestamp, to the minute."""
    return _local(dt).time().replace(second=0, microsecond=0)


def _insert(conn, table, cols, returning="id"):
    query = sql.SQL("INSERT INTO {} ({}) VALUES ({}) RETURNING {}").format(
        sql.Identifier(table),
        sql.SQL(", ").join(map(sql.Identifier, cols)),
        sql.SQL(", ").join([sql.Placeholder()] * len(cols)),
        sql.Identifier(returning),
    )
    return conn.execute(query, list(cols.values())).fetchone()[returning]


def _update(conn, table, cols, where, params):
    """UPDATE only the given columns. Column names come from this file, never from requests."""
    if not cols:
        return
    sets = sql.SQL(", ").join(sql.SQL("{} = {}").format(sql.Identifier(c), sql.Placeholder()) for c in cols)
    conn.execute(sql.SQL("UPDATE {} SET {} WHERE ").format(sql.Identifier(table), sets) + sql.SQL(where),
                 [*cols.values(), *params])

class Course:
    SELECT = "SELECT id, slug, course_name, color, start_time, end_time, day_one, day_two FROM courses"
    COLUMNS = (("name", "course_name"), ("color", "color"), ("startTime", "start_time"), ("endTime", "end_time"))

    @staticmethod
    def to_json(r):
        return {
            "id": r["slug"],
            "name": r["course_name"],
            "color": r["color"],
            "startTime": _hhmm(r["start_time"]),
            "endTime": _hhmm(r["end_time"]),
            "days": [d for d in (r["day_one"], r["day_two"]) if d],
        }

    @classmethod
    def columns(cls, values):
        """API field names -> column names, for the fields that were sent."""
        cols = {col: values[key] for key, col in cls.COLUMNS if key in values}
        if "days" in values:
            days = list(values["days"]) + [None, None]
            cols["day_one"], cols["day_two"] = days[0], days[1]
        return cols

    @classmethod
    def get(cls, conn, user_id, slug):
        return conn.execute(cls.SELECT + " WHERE user_id = %s AND slug = %s", (user_id, slug)).fetchone()

    @staticmethod
    def unique_slug(conn, user_id, name):
        base = re.sub(r"[^a-z0-9]+", "-", name.lower()).strip("-")[:40].strip("-") or "class"
        taken = {r["slug"] for r in conn.execute(
            "SELECT slug FROM courses WHERE user_id = %s AND (slug = %s OR slug LIKE %s)", (user_id, base, base + "-%")
        ).fetchall()}
        slug, n = base, 2
        while slug in taken:
            slug, n = f"{base}-{n}", n + 1
        return slug

    @staticmethod
    def create(conn, cols):
        _insert(conn, "courses", cols)

    @staticmethod
    def update(conn, course_id, cols):
        _update(conn, "courses", cols, "id = %s", (course_id,))

    @staticmethod
    def delete(conn, user_id, slug):
        return conn.execute("DELETE FROM courses WHERE user_id = %s AND slug = %s", (user_id, slug)).rowcount > 0

    @staticmethod
    def resolve(conn, user_id, values):
        """Course id for an item from `classId` (slug) or `courseId` (number), or None.
        Raises ValidationError if the class doesn't exist or belongs to someone else."""
        if values.get("courseId") is not None:
            row = conn.execute("SELECT id FROM courses WHERE id = %s AND user_id = %s",
                               (values["courseId"], user_id)).fetchone()
            field = "courseId"
        elif values.get("classId") is not None:
            row = conn.execute("SELECT id FROM courses WHERE slug = %s AND user_id = %s",
                               (values["classId"], user_id)).fetchone()
            field = "classId"
        else:
            return None
        if row is None:
            raise ValidationError({field: "No class with this id."})
        return row["id"]


class Event:
    """Calendar events that are not assignments."""

    SELECT = """
        SELECT e.id, e.event_name, e.detail, e.starts_at, e.school_event, c.slug
        FROM events e LEFT JOIN courses c ON c.id = e.course_id
        WHERE e.user_id = %s AND NOT EXISTS (SELECT 1 FROM assignments a WHERE a.event_id = e.id)
    """

    @staticmethod
    def to_json(r):
        start = _local(r["starts_at"])
        return {
            "id": str(r["id"]),
            "title": r["event_name"],
            "detail": r["detail"],
            "classId": r["slug"],
            "date": start.date().isoformat(),
            "startTime": start.strftime("%H:%M"),
            "startsAt": start.isoformat(),
            "timeLabel": _time_label(r["starts_at"]),
            "schoolEvent": r["school_event"],
        }

    @classmethod
    def get(cls, conn, user_id, event_id):
        return conn.execute(cls.SELECT + " AND e.id = %s", (user_id, event_id)).fetchone()

    @classmethod
    def list(cls, conn, user_id, start=None, end=None):
        query, params = cls.SELECT, [user_id]
        if start:
            query += " AND (e.starts_at AT TIME ZONE %s)::date >= %s"
            params += [TZNAME, start]
        if end:
            query += " AND (e.starts_at AT TIME ZONE %s)::date <= %s"
            params += [TZNAME, end]
        return conn.execute(query + " ORDER BY e.starts_at, e.id", params).fetchall()

    @staticmethod
    def create(conn, user_id, course_id, title, detail, starts_at, school_event):
        return _insert(conn, "events", {
            "user_id": user_id, "course_id": course_id, "event_name": title,
            "detail": detail, "starts_at": starts_at, "school_event": school_event,
        })

    @staticmethod
    def update(conn, event_id, cols):
        _update(conn, "events", cols, "id = %s", (event_id,))

    @staticmethod
    def delete(conn, user_id, event_id):
        return conn.execute(
            """DELETE FROM events e WHERE e.id = %s AND e.user_id = %s
               AND NOT EXISTS (SELECT 1 FROM assignments a WHERE a.event_id = e.id)""",
            (event_id, user_id),
        ).rowcount > 0


class Assignment:
    """An event with a deadline: one row in events plus one in assignments."""

    SELECT = """
        SELECT e.id, e.event_name, e.detail, e.starts_at, a.status, a.days_before_alert, c.slug
        FROM assignments a
        JOIN events e ON e.id = a.event_id
        LEFT JOIN courses c ON c.id = e.course_id
        WHERE e.user_id = %s
    """

    @staticmethod
    def to_json(r, today):
        due = _local(r["starts_at"])
        return {
            "id": str(r["id"]),
            "dueDate": due.date().isoformat(),
            "dueTime": due.strftime("%H:%M"),
            "title": r["event_name"],
            "detail": r["detail"],
            "dueLabel": _due_label(r["starts_at"], today),
            "status": r["status"],
            "classId": r["slug"],
            "daysBeforeAlert": r["days_before_alert"],
        }

    @classmethod
    def get(cls, conn, user_id, assignment_id):
        return conn.execute(cls.SELECT + " AND e.id = %s", (user_id, assignment_id)).fetchone()

    @staticmethod
    def create(conn, user_id, course_id, title, detail, due_at, days_before_alert, status):
        event_id = Event.create(conn, user_id, course_id, title, detail, due_at, False)
        conn.execute("INSERT INTO assignments (event_id, days_before_alert, status) VALUES (%s, %s, %s)",
                     (event_id, days_before_alert, status))
        return event_id

    @staticmethod
    def update(conn, assignment_id, event_cols, assignment_cols):
        _update(conn, "events", event_cols, "id = %s", (assignment_id,))
        _update(conn, "assignments", assignment_cols, "event_id = %s", (assignment_id,))

    @staticmethod
    def delete(conn, user_id, assignment_id):
        return conn.execute(
            """DELETE FROM events e USING assignments a
               WHERE a.event_id = e.id AND e.id = %s AND e.user_id = %s""",
            (assignment_id, user_id),
        ).rowcount > 0

    @classmethod
    def with_active_reminder(cls, conn, user_id, day):
        """Unfinished assignments due between `day` and `day + daysBeforeAlert`."""
        return conn.execute(
            cls.SELECT + """
              AND a.status <> 'Completed'
              AND (e.starts_at AT TIME ZONE %s)::date BETWEEN %s AND %s + a.days_before_alert
            ORDER BY e.starts_at, e.id""",
            (user_id, TZNAME, day, day),
        ).fetchall()


class Note:
    SELECT = """
        SELECT n.id, n.title, n.contents, n.pinned, n.updated_at, c.slug
        FROM notes n LEFT JOIN courses c ON c.id = n.course_id
        WHERE n.user_id = %s
    """

    @staticmethod
    def to_json(r):
        return {
            "id": str(r["id"]),
            "title": r["title"],
            "contents": r["contents"],
            "pinned": r["pinned"],
            "classId": r["slug"],
            "updatedAt": r["updated_at"].isoformat(),
            "updatedLabel": _updated_label(r["updated_at"]),
        }

    @classmethod
    def get(cls, conn, user_id, note_id):
        return conn.execute(cls.SELECT + " AND n.id = %s", (user_id, note_id)).fetchone()

    @staticmethod
    def exists(conn, user_id, note_id):
        return conn.execute("SELECT 1 FROM notes WHERE id = %s AND user_id = %s",
                            (note_id, user_id)).fetchone() is not None

    @staticmethod
    def create(conn, user_id, course_id, title, contents, pinned):
        return _insert(conn, "notes", {"user_id": user_id, "course_id": course_id, "title": title,
                                       "contents": contents, "pinned": pinned})

    @staticmethod
    def update(conn, note_id, cols):
        """Update the given columns and stamp updated_at."""
        sets = [sql.SQL("updated_at = now()")]

                     [*cols.values(), note_id])

"""AuthService: accounts, passwords and API tokens (login sequence diagram)."""
from psycopg import errors as pg_errors
from werkzeug.security import check_password_hash, generate_password_hash

import queries
from api.errors import AlreadyExists, InvalidCredentials
from api.models import Token, User
from db import pool

# Checked when the email is unknown, so a wrong email takes as long as a wrong password.
_TIMING_DUMMY = generate_password_hash("not-a-real-password")


class AuthService:
    def register(self, name, email, password):
        """Create the account and its first token. Returns (user_id, token)."""
        try:
            with pool.connection() as conn:
                if User.email_taken(conn, email):
                    raise AlreadyExists("email_taken", "An account with this email already exists.")
                user_id = User.create(conn, name, email, generate_password_hash(password))
                return user_id, Token.create(conn, user_id)
        except pg_errors.UniqueViolation:  # two sign-ups with the same email at the same moment
            raise AlreadyExists("email_taken", "An account with this email already exists.") from None

    def login(self, email, password):
        """Check the password and issue a new token. Returns (user_id, token)."""
        with pool.connection() as conn:
            row = User.find_login(conn, email)
        if row is None:
            check_password_hash(_TIMING_DUMMY, password)
            raise InvalidCredentials()
        if not check_password_hash(row["user_password_hash"], password):
            raise InvalidCredentials()
        with pool.connection() as conn:
            return row["id"], Token.create(conn, row["id"])

    def logout(self, token):
        with pool.connection() as conn:
            Token.delete(conn, token)

    def user_for_token(self, token):
        with pool.connection() as conn:
            return Token.user_for(conn, token)

    def get_user(self, user_id):
        return queries.get_user(user_id)

def _title_from(contents):
    first_line = next((line.strip() for line in contents.splitlines() if line.strip()), "")
    return first_line[:150] or "Untitled note"


class NoteRouter:
    def list_notes(self, user_id, limit=None, class_id=None):
        """Newest first (queries.get_notes), optionally one class only."""
        items = queries.get_notes(user_id, limit)
        if class_id:
            items = [n for n in items if n["classId"] == class_id]
        return items

            if not Note.delete(conn, user_id, note_id):

"""ReminderService: which assignments need a reminder on a given day.

A reminder is active from `daysBeforeAlert` days before the due date until the
due date itself, for assignments that aren't Completed.
"""
from datetime import datetime, timedelta

from api.models import Assignment
from db import pool
from queries import TZ, _local


class ReminderService:
    def due_reminders(self, user_id, day=None):
        today = day or datetime.now(TZ).date()
        with pool.connection() as conn:
            rows = Assignment.with_active_reminder(conn, user_id, today)
        result = []
        for r in rows:
            item = Assignment.to_json(r, today)
            due = _local(r["starts_at"]).date()
            item["alertDate"] = (due - timedelta(days=r["days_before_alert"])).isoformat()
            item["daysLeft"] = (due - today).days
            result.append(item)
        return result

DEFAULT_DUE_TIME = time(23, 59)


def _check_times(start, end):
    if start is not None and end is not None and end <= start:
        raise ValidationError({"endTime": "Must be after startTime."})


class ScheduleService:
    # ---------- classes ----------
    def list_classes(self, user_id):
        return queries.get_classes(user_id)

    def get_class(self, user_id, slug):
        with pool.connection() as conn:
            row = Course.get(conn, user_id, slug)
        if row is None:
            raise NotFound("Class")
        return Course.to_json(row)

    def create_class(self, user_id, values):
        _check_times(values.get("startTime"), values.get("endTime"))
        cols = Course.columns(values)
        try:
            with pool.connection() as conn:
                cols.update(user_id=user_id, slug=Course.unique_slug(conn, user_id, values["name"]))
                Course.create(conn, cols)
                return Course.to_json(Course.get(conn, user_id, cols["slug"]))
        except pg_errors.UniqueViolation:  # same name added twice at the same moment
            raise ValidationError({"name": "A class with this name was just added. Try again."}) from None

    def update_class(self, user_id, slug, values):
        """The id (slug) stays the same after a rename, so links to it keep working."""
        with pool.connection() as conn:
            row = Course.get(conn, user_id, slug)
            if row is None:
                raise NotFound("Class")
            _check_times(values.get("startTime", row["start_time"]), values.get("endTime", row["end_time"]))
            Course.update(conn, row["id"], Course.columns(values))
            return Course.to_json(Course.get(conn, user_id, slug))

    def delete_class(self, user_id, slug):
        """The class's events, assignments and notes are kept, with no class."""
        with pool.connection() as conn:
            if not Course.delete(conn, user_id, slug):
                raise NotFound("Class")

    # ---------- calendar events ----------
    def list_events(self, user_id, start=None, end=None):
        if start and end and end < start:
            raise ValidationError({"to": "Must be on or after 'from'."})
        with pool.connection() as conn:
            return [Event.to_json(r) for r in Event.list(conn, user_id, start, end)]

    def get_event(self, user_id, event_id):
        with pool.connection() as conn:
            row = Event.get(conn, user_id, event_id)
        if row is None:
            raise NotFound("Event")
        return Event.to_json(row)

    def create_event(self, user_id, values):
        with pool.connection() as conn:
            event_id = Event.create(
                conn, user_id, Course.resolve(conn, user_id, values), values["title"], values.get("detail", ""),
                at_local(values["date"], values["startTime"]), values.get("schoolEvent", False),
            )
            return Event.to_json(Event.get(conn, user_id, event_id))

    def update_event(self, user_id, event_id, values):
        with pool.connection() as conn:
            row = Event.get(conn, user_id, event_id)
            if row is None:
                raise NotFound("Event")
            cols = {}
            for key, col in (("title", "event_name"), ("detail", "detail"), ("schoolEvent", "school_event")):
                if key in values:
                    cols[col] = values[key]
            if changes_class(values):
                cols["course_id"] = Course.resolve(conn, user_id, values)
            if "date" in values or "startTime" in values:
                cols["starts_at"] = at_local(values.get("date", row["starts_at"].astimezone(TZ).date()),
                                             values.get("startTime", clock(row["starts_at"])))
            Event.update(conn, event_id, cols)
            return Event.to_json(Event.get(conn, user_id, event_id))

    def delete_event(self, user_id, event_id):
        with pool.connection() as conn:
            if not Event.delete(conn, user_id, event_id):
                raise NotFound("Event")

    def list_assignments(self, user_id, day=None):
        return queries.get_assignments(user_id, day)

    def get_assignment(self, user_id, assignment_id):
        with pool.connection() as conn:
            row = Assignment.get(conn, user_id, assignment_id)
        if row is None:
            raise NotFound("Assignment")
        return Assignment.to_json(row, datetime.now(TZ).date())

    def create_assignment(self, user_id, values):
        with pool.connection() as conn:
            assignment_id = Assignment.create(
                conn, user_id, Course.resolve(conn, user_id, values), values["title"], values.get("detail", ""),
                at_local(values["dueDate"], values.get("dueTime", DEFAULT_DUE_TIME)),
                values.get("daysBeforeAlert", 1), values.get("status", "Not started"),
            )
            return Assignment.to_json(Assignment.get(conn, user_id, assignment_id), datetime.now(TZ).date())

    def update_assignment(self, user_id, assignment_id, values):
        """Change any field. Sending only dueTime keeps the date, and vice versa."""
        with pool.connection() as conn:
            row = Assignment.get(conn, user_id, assignment_id)
            if row is None:
                raise NotFound("Assignment")
            event_cols, assignment_cols = {}, {}
            for key, col in (("title", "event_name"), ("detail", "detail")):
                if key in values:
                    event_cols[col] = values[key]
            if changes_class(values):
                event_cols["course_id"] = Course.resolve(conn, user_id, values)
            if "dueDate" in values or "dueTime" in values:
                event_cols["starts_at"] = at_local(values.get("dueDate", row["starts_at"].astimezone(TZ).date()),
                                                   values.get("dueTime", clock(row["starts_at"])))
            for key, col in (("status", "status"), ("daysBeforeAlert", "days_before_alert")):
                if key in values:
                    assignment_cols[col] = values[key]
            Assignment.update(conn, assignment_id, event_cols, assignment_cols)
            return Assignment.to_json(Assignment.get(conn, user_id, assignment_id), datetime.now(TZ).date())

    def delete_assignment(self, user_id, assignment_id):

async function apiCall(method, path, body) {
  const res = await fetch(path, {
    method,
    headers: {
      Authorization: `Bearer ${localStorage.getItem("cc_token") || "abc123"}`,
      "Content-Type": "application/json",
    },
    body: body ? JSON.stringify(body) : undefined,
  });
  if (!res.ok) throw new Error(`${method} ${path}: ${res.status}`);
  return res.json();
}

// Class sessions are generated from each course's meeting days, "Generate class sessions for the term".
function classSessionsOn(dateStr) {
  const day = DAY_NAMES[parse(dateStr).getDay()];
  return courses
    .filter(c => c.courseDays.includes(day))
    .map(c => ({
      eventID: `cls-${c.courseID}-${dateStr}`, name: `${c.short} lecture`, date: dateStr,
      start: c.courseTimeStart, end: c.courseTimeEnd, courseID: c.courseID, schoolEvent: false, isClass: true,
    }));
}

// Everything that appears on a given day: class sessions, events, and assignment due dates.
function itemsOn(dateStr) {
    const visible = item => !item.courseID || !state.hiddenCourses.has(item.courseID);
    const evs = [...classSessionsOn(dateStr), ...events.filter(e => e.date === dateStr)]
      .filter(visible)
      .map(e => ({ ...e, kind: e.isClass ? "class" : "event" }));
    const dues = assignments
      .filter(a => a.date === dateStr && visible(a))
      .map(a => ({ ...a, kind: "due", start: "23:59", name: `${a.name} due` }));
    return [...evs, ...dues].sort((a, b) => (a.start || "00:00").localeCompare(b.start || "00:00"));
  }

// NoteRouter logic: file a note into the class whose keywords appear most often.
function routeNote(text) {
  const words = text.toLowerCase();
  let best = null, bestHits = 0, bestWord = "";
  for (const c of courses) {
    for (const kw of c.keywords) {
      const hits = words.split(kw).length - 1;
      if (hits > bestHits) { best = c; bestHits = hits; bestWord = kw; }
    }
  }
  return best ? { courseID: best.courseID, keyword: bestWord } : { courseID: null, keyword: "" };
}

function weekStart(d) { return addDays(d, -((d.getDay() + 6) % 7)); } // Monday

function weekTitle(d) {
  const s = weekStart(d), e = addDays(s, 6);
  return `${MONTHS[s.getMonth()].slice(0, 3)} ${s.getDate()} – ${MONTHS[e.getMonth()].slice(0, 3)} ${e.getDate()}, ${e.getFullYear()}`;
}

function monthGrid() {
  const first = state.calMonth;
  const start = weekStart(first);
  const lastDay = new Date(first.getFullYear(), first.getMonth() + 1, 0);
  const weeks = Math.ceil((daysBetween(iso(start), iso(lastDay)) + 1) / 7);
  let cells = "";
  for (let i = 0; i < weeks * 7; i++) {
    const d = addDays(start, i);
    const ds = iso(d);
    const items = itemsOn(ds);
    const cls = ["day",
      d.getMonth() !== first.getMonth() ? "other" : "",
      ds === iso(TODAY) ? "today" : "",
      ds === state.selectedDate ? "selected" : "",
      items.length ? "has-ev" : ""].join(" ");
    cells += `<button type="button" class="${cls}" data-date="${ds}" aria-label="${ds}, ${items.length} items">
      <span class="num">${d.getDate()}</span>
      ${items.slice(0, 3).map(it => `<span class="ev pill-${colorOf(it.courseID, it.schoolEvent)}">${esc(it.name)}</span>`).join("")}
      ${items.length > 3 ? `<span class="small muted">+${items.length - 3} more</span>` : ""}
    </button>`;
  }
  return `<div class="month">
    <div class="month-head">${["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"].map(n => `<div>${n}</div>`).join("")}</div>
    <div class="month-body">${cells}</div>
  </div>`;
}

function weekGrid() {
    const s = weekStart(parse(state.selectedDate));
    let cols = "";
    for (let i = 0; i < 7; i++) {
      const d = addDays(s, i), ds = iso(d);
      cols += `<div class="week-col ${ds === iso(TODAY) ? "today" : ""}">
        <h3>${DAY_NAMES[d.getDay()]} ${d.getDate()}</h3>
        ${itemsOn(ds).map(it => `<button type="button" class="week-ev pill-${colorOf(it.courseID, it.schoolEvent)}" style="border:0;text-align:left" data-open-item="${it.kind}:${it.kind === "due" ? it.assignmentID : it.eventID}">
          <strong>${esc(it.name)}</strong>${it.kind === "due" ? "11:59 PM" : `${fmtTime(it.start)} – ${fmtTime(it.end)}`}
        </button>`).join("") || `<span class="small muted">Free</span>`}
      </div>`;
    }
    return `<div class="week">${cols}</div>`;
}

  function saveEvent(e) {
    e.preventDefault();
    const f = e.target;
    const name = $("#ev-name").value.trim();
    const date = $("#ev-date").value;
    const start = $("#ev-start").value, end = $("#ev-end").value;
    if (!name || !date) { $("#ev-error").textContent = "Every event needs a name and a date."; return; }
    if (start && end && end <= start) { $("#ev-error").textContent = "The end time must be after the start time."; return; }
    const data = { name, date, start, end, courseID: Number($("#ev-course").value) || null, schoolEvent: $("#ev-school").checked };
    const id = Number(f.dataset.id);
    if (id) Object.assign(events.find(x => x.eventID === id), data);
    else events.push({ eventID: nextId++, ...data });
    state.selectedDate = date;
    state.calMonth = new Date(parse(date).getFullYear(), parse(date).getMonth(), 1);
    closeDialog(f);
    toast(id ? "Event updated" : "Event added to your calendar");
    render();
}

async function saveAssignment(e) {
  e.preventDefault();
  const f = e.target;
  const name = $("#as-name").value.trim();
  const date = $("#as-date").value;
  const days = Number($("#as-days").value);
  if (!name || !date) { $("#as-error").textContent = "Every assignment needs a name and a due date."; return; }
  if (!(days >= 0)) { $("#as-error").textContent = "Days before alert must be 0 or more."; return; }
  const data = { name, date, courseID: Number($("#as-course").value) || null, eventID: Number($("#as-event").value) || null, daysBeforeAlert: days, schoolEvent: $("#as-school").checked };
  const payload = { title: name, dueDate: date, courseId: data.courseID, daysBeforeAlert: days };
  const id = Number(f.dataset.id);
  try {
    if (id) {
      await apiCall("PATCH", `/api/assignments/${id}`, payload);
      Object.assign(assignments.find(x => x.assignmentID === id), data);
    } else {
      const created = await apiCall("POST", "/api/assignments", payload);
      assignments.push({ assignmentID: Number(created.id), detail: "", status: "Not started", ...data });
    }
  } catch (err) {
    $("#as-error").textContent = "Couldn't save to the database. Is the server running?";
    return;
  }
  closeDialog(f);
  toast(id ? "Assignment updated" : "Assignment added");
  render();
}

async function saveNote(e) {
  e.preventDefault();
  const f = e.target;
  const contents = $("#nt-contents").value.trim();
  if (!contents) { $("#nt-error").textContent = "Write something before saving."; return; }
  let title = $("#nt-title").value.trim() || contents.split("\n")[0].slice(0, 40);
  let courseValue = $("#nt-course").value;
  let routedMsg = "";
  if (courseValue === "auto") {
    const r = routeNote(`${title} ${contents}`);
    courseValue = r.courseID || "";
    routedMsg = r.courseID ? ` · filed in ${course(r.courseID).short} (keyword: ${r.keyword})` : " · saved as a quick thought";
  }
  const data = { title, contents, courseID: Number(courseValue) || null, eventID: Number($("#nt-event").value) || null, pinned: $("#nt-pinned").checked, updated: iso(TODAY) };
  const payload = { title, contents, courseId: data.courseID, pinned: data.pinned };
  const id = Number(f.dataset.id);
  try {
    if (id) {
      await apiCall("PATCH", `/api/notes/${id}`, payload);
      Object.assign(notes.find(x => x.noteID === id), data);
    } else {
      const created = await apiCall("POST", "/api/notes", payload);
      notes.push({ noteID: Number(created.id), ...data });
    }
  } catch (err) {
    $("#nt-error").textContent = "Couldn't save to the database. Is the server running?";
    return;
  }
  closeDialog(f);
  toast(`Note saved${routedMsg}`);
  render();
}

// Deleting always asks for confirmation (from the Notes use case diagram).
function confirmDelete(button, label, onYes) {
  if (button.dataset.confirm === "1") { onYes(); return; }
  button.dataset.confirm = "1";
  const original = button.textContent;
  button.textContent = `Click again to delete this ${label}`;
  setTimeout(() => { button.dataset.confirm = ""; button.textContent = original; }, 3000);
}

/* Routing and event handling */

function render() {
    saveAll();
    renderSidebar();
    switch (state.route) {
      case "calendar": renderCalendar(); break;
      case "assignments": renderAssignments(); break;
      case "notes": renderNotes(); break;
      case "events": renderComingSoon("Campus events"); break;
      case "resources": renderComingSoon("Resources"); break;
      default: renderHome();
    }
}
  
function onRoute() {
    const route = location.hash.replace("#", "") || "home";
    state.route = ["home", "calendar", "assignments", "notes", "events", "resources"].includes(route) ? route : "home";
    render();
    window.scrollTo(0, 0);
}

document.addEventListener("click", e => {
    const t = e.target.closest("button, [data-open-note], [data-open-item], [data-folder]");
    if (!t) return;
  
    if (t.matches("[data-close]")) { closeDialog(t); return; }
  
    const action = t.dataset.action;
    if (action === "quick-note") return openNoteDialog(null, { quick: true });
    if (action === "new-note") return openNoteDialog(null, { courseID: Number(state.notesFolder) || null });
    if (action === "new-event") return openEventDialog(null, state.selectedDate);
    if (action === "new-assignment") return openAssignmentDialog();
  
    if (t.dataset.folder) { state.notesFolder = t.dataset.folder; location.hash = "#notes"; if (state.route === "notes") render(); return; }
    if (t.dataset.notesFolder) { state.notesFolder = t.dataset.notesFolder; return render(); }
    if (t.dataset.openNote) return openNoteDialog(Number(t.dataset.openNote));
    if (t.dataset.editAssignment) return openAssignmentDialog(Number(t.dataset.editAssignment));
  
    if (t.dataset.openItem) {
      const [kind, id] = t.dataset.openItem.split(":");
      if (kind === "due") return openAssignmentDialog(Number(id));
      if (kind === "class") return toast("Class sessions come from your class schedule.");
      return openEventDialog(Number(id));
}

if (t.dataset.date) {
    state.selectedDate = t.dataset.date;
    const d = parse(t.dataset.date);
    if (d.getMonth() !== state.calMonth.getMonth()) state.calMonth = new Date(d.getFullYear(), d.getMonth(), 1);
    return render();
  }

  if (t.dataset.view) { state.calView = t.dataset.view; return render(); }

  if (t.dataset.chip) {
    const id = Number(t.dataset.chip);
    state.hiddenCourses.has(id) ? state.hiddenCourses.delete(id) : state.hiddenCourses.add(id);
    return render();
  }

  if (t.dataset.cal) {
    const step = t.dataset.cal;
    if (step === "today") {
      state.selectedDate = iso(TODAY);
      state.calMonth = new Date(TODAY.getFullYear(), TODAY.getMonth(), 1);
    } else if (state.calView === "month") {
      const m = state.calMonth;
      state.calMonth = new Date(m.getFullYear(), m.getMonth() + (step === "next" ? 1 : -1), 1);
      state.selectedDate = iso(state.calMonth);
    } else {
      const d = addDays(parse(state.selectedDate), step === "next" ? 7 : -7);
      state.selectedDate = iso(d);
      state.calMonth = new Date(d.getFullYear(), d.getMonth(), 1);
    }
    return render();
  }

  if (t.id === "reminders-btn") {
    const r = activeReminders();
    return toast(r.length ? `Reminders: ${r.map(a => `${a.name} (${fmtDue(a.date).toLowerCase()})`).join(", ")}` : "No reminders right now");
  }
  if (t.id === "add-class-btn") return toast("Class setup is coming next. See the Class setup mockup.");

  if (t.id === "ev-delete") {
    return confirmDelete(t, "event", () => {
      const id = Number($("#event-form").dataset.id);
      events = events.filter(x => x.eventID !== id);
      assignments.forEach(a => { if (a.eventID === id) a.eventID = null; });
      notes.forEach(n => { if (n.eventID === id) n.eventID = null; });
      closeDialog(t); toast("Event deleted"); render();
    });
  }
  if (t.id === "as-delete") {
    return confirmDelete(t, "assignment", async () => {
      const id = Number($("#assignment-form").dataset.id);
      try { await apiCall("DELETE", `/api/assignments/${id}`); }
      catch (err) { return toast("Couldn't delete the assignment."); }
      assignments = assignments.filter(x => x.assignmentID !== id);
      closeDialog(t); toast("Assignment deleted"); render();
    });
  }
  if (t.id === "nt-delete") {
    return confirmDelete(t, "note", async () => {
      const id = Number($("#note-form").dataset.id);
      try { await apiCall("DELETE", `/api/notes/${id}`); }
      catch (err) { return toast("Couldn't delete the note."); }
      notes = notes.filter(x => x.noteID !== id);
      closeDialog(t); toast("Note deleted"); render();
    });
  }
});

  const [statusFilter, setStatusFilter] = useState('all');

  const [showForm, setShowForm] = useState(false);
  const [editingAssignment, setEditingAssignment] = useState(null);
  const addButtonRef = useRef(null);

  function closeForm() {
    setShowForm(false);
    setEditingAssignment(null);
    addButtonRef.current.focus();
    }
  async function saveAssignment(assignment) {
    const editing = Boolean(editingAssignment);
    try {
      if (editing) await onEdit(editingAssignment.id, assignment);
      else await onAdd(assignment);
    } catch {
      setStatusMessage(`Couldn't save ${assignment.title}. Please try again.`);
      return;
    }
    resetFilters();
    setStatusMessage(`${assignment.title} ${editing ? 'updated' : 'added'}. Filters reset to show all assignments.`);
    closeForm();
  }

  async function changeStatus(assignment, status) {
    try {
      await onStatusChange(assignment.id, status);
    } catch {
      setStatusMessage(`Couldn't update ${assignment.title}. Please try again.`);
      return;
    }
    setStatusMessage(`${assignment.title}: ${status}.${
      statusFilter !== 'all' && statusFilter !== status ? ' Hidden by the current status filter.' : ''
    }`);
  }

  const visibleAssignments = assignments.filter((assignment) => (
    (classFilter === 'all' || assignment.classId === classFilter)
    && (statusFilter === 'all' || assignment.status === statusFilter)
  ));
  const counts = [

    setClassFilter('all');

import { useState } from 'react';

export default function AssignmentForm({ classes, assignment, onSave, onCancel }) {
  const [titleError, setTitleError] = useState('');

  function handleSubmit(event) {
    event.preventDefault();
    const values = new FormData(event.currentTarget);
    const title = values.get('title').trim();
    if (!title) {
      setTitleError('Enter a title with at least one non-space character.');
      event.currentTarget.elements.namedItem('title').focus();
      return;
    }
    onSave({
      title,
      detail: values.get('detail').trim(),
      classId: values.get('classId'),
      dueDate: values.get('dueDate'),
    });
  }

  return (
    <form className="assignment-form" onSubmit={handleSubmit} aria-labelledby="new-assignment-heading">
      <h3 id="new-assignment-heading">{assignment ? 'Edit assignment' : 'New assignment'}</h3>
      <p className="muted">Required: title, class, and due date. Changes reset on refresh.</p>
      <label>
        Title
        <input
          name="title"
          defaultValue={assignment?.title ?? ''}
          required
          maxLength={100}
          autoFocus
          aria-invalid={Boolean(titleError)}
          aria-describedby={titleError ? 'title-error' : undefined}
          onChange={() => setTitleError('')}
        />
      </label>
      {titleError && <p className="form-error" id="title-error" role="alert">{titleError}</p>}
      <label>
        Description (optional)
        <textarea name="detail" maxLength={500} rows={3} defaultValue={assignment?.detail ?? ''} />
      </label>
      <div className="assignment-form-row">
        <label>
          Class
          <select name="classId" required defaultValue={assignment?.classId ?? ''}>
            <option value="" disabled>Choose a class</option>
            {classes.map((course) => <option key={course.id} value={course.id}>{course.name}</option>)}
          </select>
        </label>
        <label>
          Due date
          <input name="dueDate" type="date" required min="0001-01-01" max="9999-12-31" defaultValue={assignment?.dueDate ?? ''} />
        </label>
      </div>
      <div className="form-actions">
        <button className="filter-reset" type="submit">{assignment ? 'Save changes' : 'Add assignment'}</button>
        <button className="filter-reset" type="button" onClick={onCancel}>Cancel</button>
      </div>
    </form>
  );
}

const BASE = import.meta.env.VITE_API_URL ?? '';

// Dev only: seed.sql creates the token "abc123". Replace with real login later.
const getToken = () => localStorage.getItem('cc_token') ?? 'abc123';

async function get(path) {
  const res = await fetch(`${BASE}${path}`, { headers: { Authorization: `Bearer ${getToken()}` } });
  if (!res.ok) throw new Error(`${path} failed: ${res.status}`);
  return res.json();
}

async function send(method, path, body) {
  const res = await fetch(`${BASE}${path}`, {
    method,
    headers: { Authorization: `Bearer ${getToken()}`, 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  });
  if (!res.ok) throw new Error(`${method} ${path} failed: ${res.status}`);

export function useHomeData(date) {
  const [data, setData] = useState(null);
  const [error, setError] = useState(null);
  const [tick, setTick] = useState(0);
  useEffect(() => {
    let cancelled = false;
    fetchHome(date).then((d) => !cancelled && setData(d)).catch((e) => !cancelled && setError(e));
    return () => { cancelled = true; };
  }, [date, tick]);
  return { data, error, reload: () => setTick((t) => t + 1) };
}

# Runs the whole Campus Connect app in Docker: database + backend + frontend.
#
#   cd code
#   docker compose up --build        # first time (or after code changes)
#   docker compose up                # next times
#   docker compose down              # stop (data is kept)
#   docker compose down -v           # stop and delete the database data
#
# Then open http://localhost:5173 (React app) or http://localhost:5001/api/...
# See docker/README.md for details.
name: campus-connect

services:
  db:
    image: postgres:16-alpine
    environment:
      POSTGRES_PASSWORD: password   # local development only
    ports:
      - "5433:5432"                 # 5433 on your computer, so it never clashes with a locally installed PostgreSQL
    volumes:
      - db-data:/var/lib/postgresql/data
      - ./db:/schema:ro                                 # code/db/schema.sql and seed.sql
      - ./docker/initdb:/docker-entrypoint-initdb.d:ro  # loads them the first time the database starts
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres"]
      interval: 2s
      timeout: 3s
      retries: 30

  backend:
    build:
      context: .
      dockerfile: docker/backend.Dockerfile
    environment:
      DATABASE_URL: postgresql://postgres:password@db:5432/postgres
      APP_TIMEZONE: America/Denver
    depends_on:
      db:
        condition: service_healthy
    ports:
      - "5001:5001"   # Flask + API
      - "5173:5173"   # React app (the frontend container shares this one's network, see below)

  frontend:
    build:
      context: .
      dockerfile: docker/frontend.Dockerfile
    # Share the backend's network so the frontend's proxy to localhost:5001 reaches Flask,
    # exactly as it does when both run directly on one computer (vite.config.js unchanged).
    network_mode: "service:backend"
    depends_on:
      - backend

volumes:
  db-data:

#!/bin/sh
# Runs once, the first time the database container starts with an empty volume.
# Loads the project's tables and sample data from code/db/.
#
# Not using ON_ERROR_STOP on purpose: schema.sql inserts the dev token before the
# sample user exists, which prints one harmless error; seed.sql then adds it properly.

psql -U "$POSTGRES_USER" -d "$POSTGRES_DB" -f /schema/schema.sql
psql -U "$POSTGRES_USER" -d "$POSTGRES_DB" -f /schema/seed.sql
echo "Campus Connect tables and sample data loaded"
